typedef struct stackNode {  
    char data;  
    struct stackNode* next;  
}sstack;  
  
sstack* Initstack() {  
    sstack* s=new sstack;  
    s->next=nullptr;  
    return s;  
}  
  
void Push(sstack* s,BTNode* k) {  
    sstack* p=new sstack;  
    p->data=k;  
    p->next=s->next;  
    s->next=p;  
}  
  
int Empty(sstack* s) {  
    if (s->next==nullptr) return 1;  
    return 0;  
}  
  
BTNode* Pop(sstack* s) {  
    if (Empty(s)==1) {  
        cout<<"栈空"<<endl;  
        return nullptr;  
    }  
    sstack* p=s->next;  
    BTNode* k=p->data;  
    s->next=p->next;  
    delete p;  
    p=nullptr;  
    return k;  
}  
  
//获得栈顶元素  
BTNode* Get(sstack* s) {  
    if (Empty(s)==1) {  
        cout<<"栈空"<<endl;  
        return nullptr;  
    }  
    BTNode* k=s->next->data;  
    return k;  
}
