//链式队列
typedef struct QNode {
	char data;
	struct QNode* next;
}QNode;

typedef struct Queue {
	QNode* f; //队首
	QNode* r; //队尾
}Queue;

Queue Initqueue() {
	Queue q;
	q.f = new QNode;
	q.f->next = NULL;
	q.r = q.f;
	return q;
}

void enqueue(Queue* q, char x) {
	QNode* s = new QNode;
	s->data = x;
	s->next = NULL;
	q->r->next = s;
	q->r = s;
}

int empty(Queue* q) {
	if (q->f->next == NULL) return 1;   //空
	else return 0;                     //非空
}

char dequeue(Queue* q) {
	char x = ' ';
	if (empty(q) == 1) cout << "队空" << endl;
	else {
		QNode* p = q->f->next;
		x = p->data;
		//判断尾指针是否指向首元结点
		if (q->r == q->f->next) {
			q->r = q->f;
		}
		q->f->next = p->next;
		delete p;
		p = NULL;
	}
	return x;
}
