# 顺序表

```cpp
#include<iostream>
using namespace std;
#define maxx 105

typedef struct ArrayList{
    // int data[maxx];// 存放顺序表的数据
    int *data;        // 声明一个数组，方便后续扩容
    int len;          // 表长
}Array;
//初始化，值传递
Array InitArray(Array a){
    a.data = new int[maxx];   
    // 空间申请失败判断
    // if(a.data==nullptr){
    //     cout << "空间申请失败" << endl;
    //     return a;
    // }
    a.len = 0;
    return a;
}
// 添加
void Add(Array *a,int k){
    //判满
    if(a->len == maxx){
        cout << "顺序表已满" << endl;
        return;
    }
    a->data[a->len] = k;
    a->len++;
}
// 插入
void Insert(Array *a,int i,int k){
    // 判满
    if (a->len == maxx)
    {
        cout << "顺序表已满" << endl;
        return;
    }
    a->len++;
    for (int j = a->len - 1; j > i;j--){
        a->data[j] = a->data[j - 1];
    }
    a->data[i] = k;
}
// 查找（某个值的下标）
int Find(Array *a, int k){
    for (int i = 0; i < a->len; i++){
        if(a->data[i]==k)
            return i;
    }
    return -1;
}
// 删除(删除某个值)
void Delete(Array *a, int k){
    // 判空
    // 空表一定找不到
    int i = Find(a, k);
    if(i==-1){
        cout << "该数据不存在" << endl;
        return;
    }
    for (int j = i; j < a->len - 1;j++)
    {
        a->data[j] = a->data[j + 1];
    }
    a->len--;
}
void Show(Array a){
    for (int i = 0; i < a.len;i++){
        cout << a.data[i] << " ";
    }
    cout << endl;
}
int main(){

    Array a;
    a = InitArray(a);
    int op;
    do{
        cout << "请输入操作" << endl;
        cout << "1.增加" << endl;
        cout << "2.删除" << endl;
        cout << "3.插入" << endl;
        cout << "4.展示" << endl;
        cout << "0.退出" << endl;
        cin >> op;
        if (op == 1)
        {
            int n;
            cout << "输入要插入的个数" << endl;
            cin >> n;
            for (int i = 0; i < n; i++)
            {
                int m;
                cin >> m;
                Add(&a, m);
            }
        }
        else if(op==2){
            int n;
            cout << "输入要删除的值" << endl;
            cin >> n;
            Delete(&a, n);
        }
        else if (op == 3)
        {
            int n,m;
            cout << "输入要插入的位置和值" << endl;
            cin >> n >> m;
            Insert(&a, n, m);
        }
        else if(op==4){
            Show(a);
        }
        else{
            cout << "请输入正确的值" << endl;
        }

    } while (op);

    return 0;
}
```

## 单链表

```cpp
#include<iostream>
using namespace std;

// 计算机程序中一般不会出现相同的值，这里默认输入值不重复
// 结点结构
typedef struct LNode{
    int data;
    struct LNode *next;
} LNode, *LinkList;
// 初始化
LinkList InitLink(){
    LNode *head = new LNode;
    head->next = nullptr;
    return head;
}
// 查找
LNode *FindLNode(LinkList L, int x)
{
    LNode *p = L->next; // 首元结点
    while (p != nullptr && p->data != x)
    {
        p = p->next;
    }
    if (p == nullptr)
    {
        cout << "未找到该结点" << endl;
    }
    return p;
}
// 插入
// 头插法
LinkList HeadInsert(LinkList L,int x){
    LNode *n = new LNode;
    n->data = x;
    n->next = L->next;
    L->next = n;
    return L;
}
// 尾插法
LinkList RearInsert(LinkList L,int x){
    LNode *n = new LNode;
    n->data = x;
    n->next = nullptr;
    LinkList p = L;    // 必须从表头开始
    while (p->next != nullptr)
    {
        p = p->next;
    }
    p->next = n;
    return L;
}
// 指定位置插入(某值之后)
LinkList Insert(LinkList L, int x, int nx)
{
    LinkList p = FindLNode(L, x);
    if(p!=nullptr){

        LNode *n = new LNode;
        n->data = nx;
        n->next = p->next;
        p->next = n;
    }
    return L;
}
// 删除（某值）
LinkList Delete(LinkList L, int x)
{
    LinkList p = L->next;
    LinkList pre = L;
    while (p != nullptr && p->data != x)
    {
        p = p->next;
        pre = pre->next;
    }
    if (p == nullptr)
    {
        cout << "未找到该值" << endl;
    }
    else
    {
        pre->next = p->next;
        delete p;
        p = nullptr;        // 注意：防止野指针
    }
    return L;
}
// 修改
LinkList ReviseLNode(LinkList L, int x, int nx){
    LinkList p = FindLNode(L,x);
    if (p != nullptr)
    {
        p->data = nx;
    }
    return L;
}
void Show(LinkList L){
    LinkList p = L->next;
    if(p==nullptr){
        cout << "空链表" << endl;
    }
    while (p != nullptr)
    {
        cout << p->data << " ";
        p = p->next;
    }
    cout << endl;
}
int main()
{

    LinkList L = InitLink();

    int op;
    do
    {
        cout << "请输入操作" << endl;
        cout << "1.添加" << endl;
        cout << "2.指定位置插入" << endl;
        cout << "3.指定位置删除" << endl;
        cout << "4.修改某值" << endl;
        cout << "5.展示" << endl;
        cout << "0.退出" << endl;
        cin >> op;
        if (op == 1)
        {
            int n;
            cout << "输入要添加的个数" << endl;
            cin >> n;
            for (int i = 0; i < n; i++)
            {
                int m;
                cin >> m;
                L = HeadInsert(L, m);
            }
        }

        else if (op == 2)
        {
            int n, x;
            cout << "输入要插入的位置和值" << endl;
            cin >> x >> n;
            L = Insert(L, x, n);
        }

        else if (op == 3)
        {
            int x;
            cout << "输入删除的值" << endl;
            cin >> x;
            L = Delete(L, x);
        }
        else if (op == 4)
        {
            int x, nx;
            cout << "输入修改的值和新值" << endl;
            cin >> x >> nx;
            L = ReviseLNode(L, x, nx);
        }
        else if (op == 5)
        {
            Show(L);
        }
        else if(op==0){
            break;
        }
        else
        {
            cout << "请输入正确的值" << endl;
        }

    } while (op);
    return 0;
}
```
