## 邻接矩阵
```cpp
#include<iostream>
using namespace std;
#define inf 10001   //模拟无限大
//带权无向图为例，n个点，m条边   n<=100  0<=w<=10000
int n, m;
char v[105];      //顶点
int e[105][105];  //利用二维数组存储图

int Find(char x){
    int j = -1;
    for (j = 1; j <= n; j++)
    {
        if(v[j]==x)
            break;
    }
    return j;
}
int main(){

    cin >> n >> m;
    for (int i = 1; i <= n;i++){
        cin >> v[i];
    }
    for (int i = 1; i <= n;i++){
        for (int j = 1; j <= n;j++){
            e[i][j] = inf;        //初始化
        }
    }
    char x, y;
    int xi, yi, w;
    for (int i = 1; i <= m;i++){
        cin >> x >> y >> w;
        xi = Find(x);
        yi = Find(y);
        e[xi][yi] = e[yi][xi] = w;
    }

    char f;
    int fi, d=0;
    cout << "请输入要查找的结点" << endl;
    cin >> f;
    fi = Find(f);
    for (int i = 1; i <= n;i++){
        if(e[fi][i]<inf){
            d++;
        }
    }
    cout << "该结点的度为" << d << endl;
    return 0;
}
```

## 邻接表
```cpp
#include <iostream>
using namespace std;
#define inf 10001

// 带权无向图为例，n个点，m条边   n<=100  0<=w<=10000
int n, m;

//邻接点链表的结点结构
typedef struct ENode{
    int adj;   //邻接点的下标
    int w;     //边权
    struct ENode *next;
} ENode;

//邻接点结构
struct Graph{
    char data;    //顶点数组
    ENode *first; //头指针，指向邻接点链表
} g[105];

int Find(char x)
{
    int j = -1;
    for (j = 1; j <= n; j++)
    {
        if (g[j].data == x)
            break;
    }
    return j;
}

int main(){

    cin >> n >> m;
    for (int i = 1; i <= n;i++){
        cin >> g[i].data;
        g[i].first = nullptr;
    }
    char x, y;
    int xi, yi, w;

    for (int i = 1; i <= m;i++){
        cin >> x >> y >> w;
        xi = Find(x);
        yi = Find(y);

        //开始插入
        //x------y ,y做邻接点
        ENode *s = new ENode;
        s->adj = yi;
        s->w = w;
        s->next = g[xi].first;
        g[xi].first = s;

        //y------x ,x做邻接点
        s = new ENode;
        s->adj = xi;
        s->w = w;
        s->next = g[yi].first;
        g[yi].first = s;
        // 如果是有向图，则只要连接一次，连接出度或者入度，称为邻接表和逆邻接表
    }

    char f;
    int fi, d;
    cout << "请输入要查找的结点" << endl;
    cin >> f;
    fi = Find(f);
    ENode *p = g[fi].first;
    while(p!=nullptr){
        d++;
        p=p->next;
    }
    cout << "该结点的度为" << d << endl;

    return 0;
}
```

## 十字链表
```cpp
#include <iostream>
using namespace std;
#define inf 10001

// 带权有向图为例，n个点，m条边   n<=100  0<=w<=10000
int n, m;

typedef struct ENode{
    int w;
    int tail;
    int head;
    struct ENode *tnext;
    struct ENode *hnext;
} ENode;

struct Graph{
    char data;
    ENode *firstout;
    ENode *firstin;
} g[105];

int Find(char x)
{
    int j = -1;
    for (j = 1; j <= n; j++)
    {
        if (g[j].data == x)
            break;
    }
    return j;
}

int main(){

    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> g[i].data;
        g[i].firstin = nullptr;
        g[i].firstout = nullptr;
    }
    char x, y;
    int xi, yi, w;
    for (int i = 1; i <= m; i++)
    {
        cin >> x >> y >> w;
        xi = Find(x);
        yi = Find(y);

        // 开始插入
        // x-------->y
        ENode *s = new ENode;
        s->w = w;
        s->tail = xi;
        s->head = yi;
        //s插到x的出边链表中
        s->tnext = g[xi].firstout;
        g[xi].firstout = s;
        // s插到y的入边链表中
        s->hnext = g[yi].firstin;
        g[yi].firstin = s;
    }
    
    char f;
    int fi, d;
    cout << "请输入要查找的结点" << endl;
    cin >> f;
    fi = Find(f);
    ENode *p = g[fi].firstout;
    while (p != nullptr)
    {
        d++;
        p = p->tnext;
    }
    p = g[fi].firstin;
    while (p != nullptr)
    {
        d++;
        p = p->hnext;
    }
    cout << "该结点的度为" << d << endl;

    return 0;
}
```

## 多重邻接表

```cpp
#include <iostream>
using namespace std;
#define inf 10001

// 带权无向图为例，n个点，m条边   n<=100  0<=w<=10000
int n, m;

typedef struct ENode
{
    int w;
    int x;
    int y;
    struct ENode *xnext;
    struct ENode *ynext;
} ENode;

struct Graph
{
    char data;
    ENode *first;
} g[105];

int Find(char x)
{
    int j = -1;
    for (j = 1; j <= n; j++)
    {
        if (g[j].data == x)
            break;
    }
    return j;
}
int main(){

    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> g[i].data;
        g[i].first = nullptr;
    }
    char x, y;
    int xi, yi, w;

    for (int i = 1; i <= m;i++){
        cin >> x >> y >> w;
        xi = Find(x);
        yi = Find(y);

        // 开始插入
        // x--------y
        ENode *s = new ENode;
        s->w = w;     
        //作为x的边，插到x的边链表中
        s->x = xi;
        s->xnext = g[xi].first;
        g[xi].first = s;
        //作为y的边，插到y的边链表中
        s->y = yi;
        s->ynext = g[yi].first;
        g[yi].first = s;
    }

    char f;
    int fi, d;
    cout << "请输入要查找的结点" << endl;
    cin >> f;
    fi = Find(f);
    ENode *p = g[fi].first;
    while (p != nullptr)
    {
        d++;
        if(p->x==fi){
            p = p->xnext;
        }
        else{
            p = p->ynext;
        }
    }
    cout << "该结点的度为" << d << endl;

    return 0;
}
```
