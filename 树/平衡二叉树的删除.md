
删除后(调整前)**最多有一个**结点失衡——最小失衡子树的根结点

左子树**删除**一个结点 == 右子树**增加**一个结点
右子树**删除**一个结点 == 左子树**增加**一个结点

通过上面的思想，可以在删除之后通过**旋转**来调整失衡——[[平衡二叉树（AVL树）的概念和旋转]]

*注意：删除并调整后，最小失衡子树的根节点的父节点可能会**失衡**，所以必须使用**递归**的方法，层层退回调整*

```cpp
//删除
AVLTree Del(AVLTree root,int x){
    if (root == nullptr)
    {
        cout << "空树或没有找到该结点，无法删除" << endl;
        return root;
    }
    if (x < root->data)
    {
        root->l = Del(root->l, x);
        //开始调整
        //删除左边 == 右边+1
        // RR / RL
        if(Get_h(root->r)-Get_h(root->l)>1){
            AVLNode *right = root->r;
            if(Get_h(right->r)>=Get_h(right->l)){
                //RR
                root = RRrotation(root);
            }
            else{
                //RL
                root = RLrotation(root);
            }
        }
    }
    else if (x > root->data)
    {
        root->r = Del(root->r, x);
        // 开始调整
        // 删除右边 == 左边+1
        //  LL / LR
        if (Get_h(root->l) - Get_h(root->r) > 1)
        {
            AVLNode *left = root->r;
            if (Get_h(left->l) >= Get_h(left->r))
            {
                // LL
                root = LLrotation(root);
            }
            else
            {
                // LR
                root = LRrotation(root);
            }
        }
    }
    else
    {
        if(root->l!=nullptr && root->r!=nullptr){
            // 度为2
            // 找root的中序遍历的后继
            AVLNode *y = root->r;
            while(y->l!=nullptr){
                y = y->l;
            }
            root->data = y->data;
            root->r = Del(root->r, y->data);
        }
        else{
            // 度为1或0
            AVLNode *p = root;
            if(root->l!=nullptr){
                root = root->l;
            }
            else{
                root = root->r;
            }
            delete p;
            p = nullptr;
        }
    }
    // 更新高度
    if(root!=nullptr)
    root->h = maxx(Get_h(root->l), Get_h(root->r));
    return root; // 链接父结点
}
```
