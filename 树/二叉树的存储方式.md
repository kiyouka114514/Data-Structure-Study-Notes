
# **二叉树**的2种独特存储方式
试用案例
```
7
a
b a 0
c b 0
d b 1
e d 0
f d 1
g e 1
请输入要查找的节点：
d
该结点的左孩子是e
该结点的右孩子是f
该结点的父亲是b
```

## 一、利用完全二叉树

基于[[二叉树的概念和性质]]的性质5，二叉树可以用顺序存储方式，利用一维数组，下标来表示父子关系
普通树可以**在完全二叉树的基础上**，进行修剪，利用完全二叉树的方式存储

```cpp
#include<iostream>
using namespace std;
const int maxx = 1005;
char my_data[maxx];

int Find(char fx) {
	for (int i = 1; i <= maxx; i++) {
		if (my_data[i] == fx) return i;
	}
}
//找父亲
void FindFa(char x) {
	int i = Find(x);
	if (i == 1) cout << "该结点是根结点" << endl;
	else cout <<"该结点的父亲是"<< my_data[i / 2] << endl;
}
//找儿子
void FindSon(char x) {
	int i = Find(x);
	if (my_data[i * 2] == ' ') cout << "该结点没有左孩子" << endl;
	else cout << "该结点的左孩子是" << my_data[i * 2] << endl;
	if (my_data[i * 2 + 1] == ' ') cout << "该结点没有右孩子" << endl;
	else cout << "该结点的右孩子是" << my_data[i * 2 + 1] << endl;
}
int main() {

	//初始化
	for (int i = 1; i < maxx; i++) {
		my_data[i] = ' ';
	}

	int n;
	cin >> n;
	char r;
	cin >> r;
	my_data[1] = r;

	char x, fx;
	int flag;    //0 为左孩子，1 为右孩子

	//插入
	for (int i = 2; i <= n; i++) {
		cin >> x >> fx >> flag;
		int j = Find(fx);
		if (flag == 0) my_data[j * 2] = x;
		else my_data[j * 2 + 1] = x;
	}

	//开始查找
	char f;
	cout << "请输入要查找的节点：" << endl;
	cin >> f;
	FindSon(f);
	FindFa(f);

	return 0;
}

```
这种存储方式相当**浪费空间**，一般不用

## 二、利用**二叉链表**

### 普通版

![[Pasted image 20261025153747.png]]

| 左孩子的地址 | data | 右孩子的地址 |
| ------ | ---- | ------ |
```cpp
#include<iostream>
using namespace std;

typedef struct BTNode {
	char data;
	struct BTNode* l;  // 左孩子
	struct BTNode* r;  // 右孩子
}BTNode,*BTree;

BTree InitBTree(char r) {
	BTNode* s = new BTNode;
	s->data = r;
	s->l = NULL;
	s->r = NULL;
	return s;
}
/*
	利用递归算法：
	1.找root，找到返回自身，找不到，执行2
	2.找l，找到返回自身，找不到，执行3
	3.找r，找到返回自身，找不到，返回NULL
*/
BTNode* Find(BTree root, char x) {
	if (root->data = x) return root;

	BTNode* ans = NULL;
	if (root->l != NULL) {
		ans = Find(root->l, x);
		if (ans != NULL) return ans;
	}
	if (root->r != NULL) {
		ans = Find(root->r, x);
		if (ans != NULL) return ans;
	}
	return NULL;
}
BTree Insert(BTree root, char x, char fx, int flag) {
	BTNode* s = new BTNode;
	s->data = x;
	s->l = NULL;
	s->r = NULL;

	BTNode* p = Find(root, fx);
	if (flag == 0) p->l = s;
	else p->r = s;
	return root;
}
//找儿子
void FindSon(BTree root, char x) {
	BTNode* s = Find(root, x);

	if (s->l == NULL) cout << "该结点没有左孩子" << endl;
	else cout << "该结点的左孩子是" << s->l->data << endl;
	if (s->r == NULL) cout << "该结点没有右孩子" << endl;
	else cout << "该结点的右孩子是" << s->r->data << endl;
}
int main() {

	int n, flag;
	cin >> n;
	char r;
	cin >> r;
	BTree root = InitBTree(r);

	//插入
	char x, fx;
	for (int i = 1; i < n; i++) {
		cin >> x >> fx >> flag;
		root = Insert(root, x, fx, flag);
	}

	//开始查找
	char f;
	cout << "请输入要查找的节点：" << endl;
	cin >> f;
	FindSon(root, f);

	return 0;
}
```
显然，和[[树的存储方式]]的孩子兄弟表示法一样，查找不了父亲，所以进行改良

### 进阶版

非常简单，只需要在每个结点里再**存储父亲地址**即可
belike:
![[Pasted image 20261025230408.png]]

| 左孩子地址 | data | 右孩子地址 | 父亲地址 |
| ----- | ---- | ----- | ---- |

```cpp
#include<iostream>
using namespace std;

typedef struct BTNode {
	char data;
	struct BTNode* l;  // 左孩子
	struct BTNode* r;  // 右孩子
	struct BTNode* fa; // 父亲结点
}BTNode, * BTree;

BTree InitBTree(char r) {
	BTNode* s = new BTNode;
	s->data = r;
	s->l = NULL;
	s->r = NULL;
	s->fa = NULL;
	return s;
}
/*
	利用递归算法：
	1.找root，找到返回自身，找不到，执行2
	2.找l，找到返回自身，找不到，执行3
	3.找r，找到返回自身，找不到，返回NULL
*/
BTNode* Find(BTree root, char x) {
	if (root->data == x) return root;

	BTNode* ans = NULL;
	if (root->l != NULL) {
		ans = Find(root->l, x);
		if (ans != NULL) return ans;
	}
	if (root->r != NULL) {
		ans = Find(root->r, x);
		if (ans != NULL) return ans;
	}
	return NULL;
}
BTree Insert(BTree root, char x, char fx, int flag) {
	BTNode* s = new BTNode;
	s->data = x;
	s->l = NULL;
	s->r = NULL;
	s->fa = NULL;

	BTNode* p = Find(root, fx);
	s->fa = p;
	if (flag == 0) p->l = s;
	else p->r = s;
	return root;
}
//找儿子
void FindSon(BTree root, char x) {
	BTNode* s = Find(root, x);

	if (s->l == NULL) cout << "该结点没有左孩子" << endl;
	else cout << "该结点的左孩子是" << s->l->data << endl;
	if (s->r == NULL) cout << "该结点没有右孩子" << endl;
	else cout << "该结点的右孩子是" << s->r->data << endl;
}
//找父亲
void FindFa(BTree root, char fx) {
	BTNode* s = Find(root, fx);

	if (s->fa == NULL) cout << "该结点为根结点" << endl;
	else cout << "该结点的父亲是" << s->fa->data << endl;
}
int main() {

	int n, flag;
	cin >> n;
	char r;
	cin >> r;
	BTree root = InitBTree(r);

	//插入
	char x, fx;
	for (int i = 1; i < n; i++) {
		cin >> x >> fx >> flag;
		root = Insert(root, x, fx, flag);
	}

	//开始查找
	char f;
	cout << "请输入要查找的节点：" << endl;
	cin >> f;
	FindSon(root, f);
	FindFa(root, f);

	return 0;
}
```
