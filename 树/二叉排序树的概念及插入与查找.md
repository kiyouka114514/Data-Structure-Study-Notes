
# 概念

利用**中序遍历**
顺序性：
如果一个结点的左子树非空，那么左子树上的结点都小于该结点
如果一个结点的右子树非空，那么右子树上的结点都大于该结点

左<根<右      或        左>根>右

优点：使查找的效率降低到 **h(树的高度)** 次，n个结点的查找速度近似于$logn$（完全二叉树）[[二叉树的概念和性质]]
缺点：查找的效率**不稳定**，比如斜树的速度为n，如何优化——[[平衡二叉树（AVL树）的概念和旋转]]

# 插入与查找

```cpp
#include <iostream>
using namespace std;

typedef struct BSTNode
{
    int data;
    struct BSTNode *l;
    struct BSTNode *r;

} BSTNode, *BSTree;
BSTNode* CreateBSTNode(int x){
    BSTNode *s = new BSTNode;
    s->data = x;
    s->l = s->r = nullptr;
    return s;
}
//查找x----非递归
BSTNode* Search1(BSTree root,int x){

    BSTNode *p = root;
    while(p!=nullptr && p->data==x){
        if(x<p->data)
            p = p->l; 
        else
            p = p->r;
    }
    return p;
}
//查找x----递归
BSTNode* Search2(BSTree root,int x){

    if(root==nullptr || root->data==x)
        return root;
    if(x<root->data)
        return Search2(root->l,x);
    else
        return Search2(root->r, x);
}
//插入----非递归
BSTree Insert1(BSTree root, int x)
{
    if(root==nullptr){
        root = CreateBSTNode(x);
        return root;
    }
    BSTNode *p = root;
    BSTNode *pre = nullptr;

    BSTNode *s = CreateBSTNode(x);

    while(p!=nullptr){
        pre = p;
        if(x<p->data)
            p = p->l;
        else
            p = p->r;
    }
    if(s->data<pre->data)
        pre->l = s;
    else
        pre->r = s;

    return root;
}
//插入----递归
BSTree Insert2(BSTree root,int x){
    if (root == nullptr)
    {
        root = CreateBSTNode(x);
        return root;
    }

    if(x<root->data)
        root->l=Insert2(root->l, x);
    else
        root->r=Insert2(root->r, x);

    return root;
}
// 中序遍历
void InOrderBT(BSTree root)
{
    if (root == NULL)
        return;
    InOrderBT(root->l);
    cout << root->data << " ";
    InOrderBT(root->r);
}
int main(){
    int n, x;
    BSTree root = nullptr;
    cin >> n;
    for (int i = 1; i <= n;i++){
        cin >> x;
        root = Insert2(root, x);
    }

    InOrderBT(root);
    int f;
    cout << "请输入要查找的数" << endl;
    cin >> f;
    if(Search1(root,f))
        cout << "存在" << endl;
    else
        cout << "不存在" << endl;
    return 0;
}
```
